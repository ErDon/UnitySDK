using System;
using ErDon.UnitySDK;
using ErDon.UnitySDK.Editor;
using UnityEditor;
using UnityEngine;
[InitializeOnLoad]
public static class AnalyticsPlayChecks
{
    const string Key = "ErDon.Tests.AnalyticsPlay";
    static int phase, count, late;
    static AnalyticsResult result;
    static double deadline;
    static bool finished;
    static AnalyticsPlayChecks() { EditorApplication.playModeStateChanged += Changed; }
    public static void Run() { SessionState.SetBool(Key, true); EditorApplication.isPlaying = true; }
    static void Check(bool ok, string message) { if (!ok) throw new Exception(message); }
    static void Setup(MockAnalyticsOutcome outcome)
    {
        EditorSdkBootstrap.Initialize(MockLoginOutcome.Success, (l,s) => {}, analytics: outcome);
        result = null;
    }
    static void Report(int timeout = 1000) { SDK.ReportEvent(new AnalyticsEvent("preview"), value => { result = value; count++; }, timeout); }
    static void Changed(PlayModeStateChange state)
    {
        if (state != PlayModeStateChange.EnteredPlayMode || !SessionState.GetBool(Key, false)) return;
        try
        {
            var settings = SdkProjectSettings.instance;
            settings.platform = DistributionPlatform.Mock; settings.production = false; settings.gameId = "analytics-test";
            settings.login = settings.storage = settings.share = settings.payments = settings.rewardedAds = settings.network = false;
            settings.analytics = true;
            deadline = EditorApplication.timeSinceStartup + 15;
            Setup(MockAnalyticsOutcome.Dispatched); Report();
            EditorApplication.update += Step;
        }
        catch (Exception e) { Finish(false, e.Message); }
    }
    static void Step()
    {
        if (finished) return;
        try
        {
            if (EditorApplication.timeSinceStartup > deadline) throw new Exception("test deadline");
            if (phase == 5) { Check(late == 0 && count == 5, "shutdown suppression/counts"); Finish(true, ""); return; }
            if (result == null) return;
            Check(!string.IsNullOrEmpty(result.CallId), "missing correlation");
            switch (phase++)
            {
                case 0:
                    Check(result.Dispatched, "mock success"); Setup(MockAnalyticsOutcome.Failed); Report(); break;
                case 1:
                    Check(result.Error.Code == "ANALYTICS.PLATFORM_FAILED", "mock failure"); Setup(MockAnalyticsOutcome.Timeout); Report(50); break;
                case 2:
                    Check(result.Error.Code == "ANALYTICS.TIMEOUT", "host timeout");
                    result = null;
                    var operation = SDK.ReportEvent(new AnalyticsEvent("cancel"), value => { result = value; count++; });
                    operation.Cancel(); operation.Cancel(); break;
                case 3:
                    Check(result.Error.Code == "ANALYTICS.CANCELLED", "cancel");
                    SdkProjectSettings.instance.analytics = false; Setup(MockAnalyticsOutcome.Dispatched); Report(); break;
                case 4:
                    Check(result.Error.Code == "MODULE.DISABLED", "disabled module");
                    SdkProjectSettings.instance.analytics = true; Setup(MockAnalyticsOutcome.Dispatched);
                    SDK.ReportEvent(new AnalyticsEvent("stale"), value => late++); UnitySdkHost.Shutdown(); break;
            }
        }
        catch (Exception e) { Finish(false, e.Message); }
    }
    static void Finish(bool success, string message)
    {
        if (finished) return; finished = true; SessionState.SetBool(Key, false); EditorApplication.update -= Step;
        UnitySdkHost.Shutdown();
        if (success) Debug.Log("SDK_ANALYTICS_PLAY_PASS"); else Debug.LogError("SDK_ANALYTICS_PLAY_FAIL: " + message);
        EditorApplication.Exit(success ? 0 : 1);
    }
}
