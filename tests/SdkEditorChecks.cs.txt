using System;
using System.IO;
using System.Linq;
using ErDon.UnitySDK.Editor;
using UnityEditor.Build;
using UnityEditor.Compilation;
using UnityEngine;
public static class SdkEditorChecks {
 public static void Run() {
  var s=SdkProjectSettings.instance;
  s.gameId="test-game"; s.platform=DistributionPlatform.Mock; s.production=false;
  if(s.Validate().Count!=0) throw new Exception("valid editor mock rejected");
  s.production=true;
  if(s.Validate().Count==0) throw new Exception("production mock allowed");
  s.production=false;s.rewardedAds=true;
  if(s.Validate().Count==0) throw new Exception("unimplemented ads allowed");
  s.rewardedAds=false;s.payments=false;s.platform=DistributionPlatform.WeChat;s.appId="test-app";
  s.storage=true;s.share=true;
  var exported=s.CreateRuntimeConfiguration();
  var roundTrip=JsonUtility.FromJson<ErDon.UnitySDK.SdkRuntimeConfiguration>(JsonUtility.ToJson(exported));
  if(roundTrip.gameId!="test-game" || roundTrip.appId!="test-app" || roundTrip.platform!="WeChat"
      || !roundTrip.ToSdkConfig().StorageEnabled || !roundTrip.ToSdkConfig().ShareEnabled)
      throw new Exception("Runtime configuration JSON mapping failed");
  Debug.Log("SDK_RUNTIME_CONFIG_PASS");
  exported.gameId="changed";
  if(s.gameId!="test-game") throw new Exception("Export mutated project settings");
  if(s.Validate().Count==0) throw new Exception("missing adapter allowed");
  bool blocked=false;try {new SdkBuildGuard().OnPreprocessBuild(null);}catch(BuildFailedException){blocked=true;}
  if(!blocked) throw new Exception("build not blocked");
  const string vendorRoot="Assets/Plugins/com.tiktok.minigame";
  if(UnityEditor.AssetDatabase.IsValidFolder(vendorRoot) || Directory.Exists(vendorRoot))
      throw new Exception("Isolation fixture path already exists; refusing to modify it");
  if(!UnityEditor.AssetDatabase.IsValidFolder("Assets/Plugins"))UnityEditor.AssetDatabase.CreateFolder("Assets","Plugins");
  UnityEditor.AssetDatabase.CreateFolder("Assets/Plugins","com.tiktok.minigame");
  try {
      if(PlatformDependencyCatalog.CheckVendorInstallations("WeChat").Count==0)
          throw new Exception("Official vendor root without adapter escaped detection");
      if(PlatformDependencyCatalog.CheckVendorInstallations("TikTok").Count!=0)
          throw new Exception("Target vendor root rejected");
  } finally {UnityEditor.AssetDatabase.DeleteAsset(vendorRoot);}
  // A real Unity precompiled reference must be checked even though it is not a source assembly.
  var reference = CompilationPipeline.GetAssemblies(AssembliesType.Player)
      .SelectMany(a=>a.compiledAssemblyReferences).FirstOrDefault();
  if(reference == null) throw new Exception("No precompiled reference available for isolation test");
  var fixture = new PlatformDependency { Platform="ForeignTestPlatform",Package="test.fixture",Version="1.0.0",
      Assemblies=new[]{Path.GetFileNameWithoutExtension(reference)}, AssetRoots=new string[0] };
  if(PlatformDependencyCatalog.CheckPlayerAssemblies("WeChat",new[]{fixture}).Count==0)
      throw new Exception("Precompiled reference escaped platform check");
  s.platform=DistributionPlatform.Mock;s.SaveSettings();
  Debug.Log("SDK_EDITOR_CHECKS_PASS");
 }
}
