using System;
using System.IO;
using ErDon.UnitySDK.Editor;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEditor.SceneManagement;
using UnityEngine;
public static class MiniGameDevelopmentBuild {
 public static void ConfigureWeChat() { Configure(DistributionPlatform.WeChat,"ERDON_PLATFORM_WECHAT"); }
 public static void ConfigureDouyin() { Configure(DistributionPlatform.Douyin,"ERDON_PLATFORM_DOUYIN"); }
 public static void ConfigureTikTok() { Configure(DistributionPlatform.TikTok,"ERDON_PLATFORM_TIKTOK"); }
 static void Configure(DistributionPlatform platform,string symbol) {
  PlayerSettings.SetScriptingDefineSymbolsForGroup(BuildTargetGroup.WebGL,symbol);
  var s=SdkProjectSettings.instance;
  s.platform=platform;s.gameId="development-build-probe";s.appId="build-only-placeholder";
  s.production=false;s.login=false;s.storage=true;s.share=true;s.rewardedAds=true;s.analytics=true;
  s.network=false;s.payments=false;s.SaveSettings();
  Debug.Log("SDK_DEVELOPMENT_BUILD_CONFIGURED");
 }
 public static void ConfigureAllCapabilitiesOff() {
  var s=SdkProjectSettings.instance;
  if(s.platform==DistributionPlatform.Mock || s.production)throw new Exception("Configure an isolated non-production platform project first");
  s.login=s.storage=s.share=s.rewardedAds=s.analytics=s.network=s.payments=false;
  s.ApplyCapabilityDefines();
  Debug.Log("SDK_ALL_CAPABILITIES_OFF_CONFIGURED");
 }
 public static void ConfigureLoginOnly() {
  var s=SdkProjectSettings.instance;
  if(s.platform==DistributionPlatform.Mock || s.production)throw new Exception("Configure an isolated non-production platform project first");
  s.login=true;s.storage=s.share=s.rewardedAds=s.analytics=s.network=s.payments=false;
  s.ApplyCapabilityDefines();
  Debug.Log("SDK_LOGIN_ONLY_CONFIGURED");
 }
 public static void Build() {
  if(EditorUserBuildSettings.activeBuildTarget!=BuildTarget.WebGL)throw new Exception("Run with -buildTarget WebGL");
  var scene=EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
  var settings=SdkProjectSettings.instance;
  if(settings.Validate(true).Count!=0)throw new Exception("Invalid development configuration");
  var probe=new GameObject("SDK Build Probe").AddComponent<MiniGameBuildProbe>();
  probe.configuration=settings.CreateRuntimeConfiguration();
  const string scenePath="Assets/DevelopmentBuildProbe.unity";
  if(!EditorSceneManager.SaveScene(scene,scenePath))throw new Exception("Scene save failed");
  PlayerSettings.WebGL.compressionFormat=WebGLCompressionFormat.Disabled;
  var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions {
   scenes=new[]{scenePath},locationPathName="Builds/DevelopmentWebGL",target=BuildTarget.WebGL,
   options=BuildOptions.Development
  });
  Directory.CreateDirectory("Builds");
  File.WriteAllText("Builds/development-build-result.txt",report.summary.result+" errors="+report.summary.totalErrors);
  if(report.summary.result!=BuildResult.Succeeded)throw new Exception("Development WebGL build failed: "+report.summary.result);
  Debug.Log("SDK_DEVELOPMENT_WEBGL_BUILD_PASS");
 }
}
