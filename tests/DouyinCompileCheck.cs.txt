using UnityEditor;
using UnityEngine;
public static class DouyinCompileCheck {
 public static void Configure() {
  PlayerSettings.SetScriptingDefineSymbolsForGroup(BuildTargetGroup.Standalone,"ERDON_PLATFORM_DOUYIN");
  Debug.Log("DOUYIN_DEFINE_CONFIGURED");
 }
 public static void Verify() {
  var t=System.Type.GetType("ErDon.UnitySDK.Douyin.DouyinCredentialProvider, ErDon.UnitySDK.Douyin");
  if(t==null) throw new System.Exception("Douyin adapter not loaded");
  var storage=System.Type.GetType("ErDon.UnitySDK.Douyin.DouyinStorage, ErDon.UnitySDK.Douyin");
  if(storage==null || !typeof(ErDon.UnitySDK.IStorage).IsAssignableFrom(storage)) throw new System.Exception("Douyin storage not loaded");
  Debug.Log("DOUYIN_UNITY_COMPILE_PASS");
  var share=System.Type.GetType("ErDon.UnitySDK.Douyin.DouyinShare, ErDon.UnitySDK.Douyin");
  if(share==null || !typeof(ErDon.UnitySDK.IShare).IsAssignableFrom(share)) throw new System.Exception("Douyin share not loaded");
  Debug.Log("DOUYIN_SHARE_UNITY_COMPILE_PASS");
 }
}
