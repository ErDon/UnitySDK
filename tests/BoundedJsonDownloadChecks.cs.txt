using System;
using System.Reflection;
using System.Text;
using ErDon.UnitySDK;
using UnityEditor;
using UnityEngine;
using UnityEngine.Networking;

// Runs the actual Unity DownloadHandlerScript implementation with injected receive chunks.
public static class BoundedJsonDownloadChecks {
    static readonly Type Type = typeof(UnitySessionExchange).Assembly.GetType("ErDon.UnitySDK.BoundedJsonDownload", true);
    const BindingFlags Flags = BindingFlags.Instance | BindingFlags.NonPublic;
    static object New() { return Activator.CreateInstance(Type, true); }
    static bool Feed(object handler, byte[] data, int count) {
        return (bool)Type.GetMethod("ReceiveData", Flags).Invoke(handler, new object[]{data, count});
    }
    static string Text(object handler) { return (string)Type.GetMethod("ReadText", Flags).Invoke(handler, null); }
    static bool Exceeded(object handler) { return (bool)Type.GetProperty("Exceeded", Flags).GetValue(handler); }
    static void Check(bool condition, string message) { if (!condition) throw new Exception(message); }
    public static void Run() {
        try {
            using (var handler = (DownloadHandler)New()) {
                var json = "{\"session\":\"test-session\",\"errorCode\":\"中文\"}";
                var bytes = Encoding.UTF8.GetBytes(json);
                foreach (var value in bytes) Check(Feed(handler, new[]{value}, 1), "split UTF8 rejected");
                Check(Text(handler) == json && !Exceeded(handler), "split UTF8 response corrupted");
            }
            using (var handler = (DownloadHandler)New()) {
                var chunk = new byte[8192]; for (int i=0;i<chunk.Length;i++) chunk[i]=(byte)'x';
                for (int i=0;i<8;i++) Check(Feed(handler, chunk, chunk.Length), "exact limit rejected");
                Check(Text(handler).Length == 65536 && !Exceeded(handler), "exact limit content mismatch");
                Check(!Feed(handler, new byte[]{1}, 1) && Exceeded(handler), "overflow accepted");
                Check(!Feed(handler, new byte[0], 0), "overflow state reset");
                Check(Text(handler).Length == 65536, "overflow retained extra bytes");
            }
            using (var handler = (DownloadHandler)New()) {
                Check(!Feed(handler, new byte[65537], 65537) && Exceeded(handler), "oversize first chunk accepted");
                Check(Text(handler).Length == 0, "oversize first chunk retained");
            }
            using (var handler = (DownloadHandler)New()) {
                Check(!Feed(handler, null, 0) && !Feed(handler, new byte[1], 2) && !Feed(handler, new byte[1], -1), "invalid chunk accepted");
                Check(Text(handler).Length == 0, "invalid chunk mutated buffer");
            }
            Debug.Log("SDK_BOUNDED_JSON_DOWNLOAD_PASS"); EditorApplication.Exit(0);
        } catch(Exception ex) { Debug.LogError("SDK_BOUNDED_JSON_DOWNLOAD_FAIL: " + ex.Message); EditorApplication.Exit(1); }
    }
}
