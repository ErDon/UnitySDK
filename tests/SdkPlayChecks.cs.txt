using System;
using ErDon.UnitySDK;
using ErDon.UnitySDK.Editor;
using UnityEditor;
using UnityEngine;
[InitializeOnLoad]
public static class SdkPlayChecks
{
    const string Key="ErDon.Tests.Play";
    static int phase;
    static double deadline;
    static bool finished;
    static SdkPlayChecks() {
        EditorApplication.playModeStateChanged += Changed;
    }
    public static void Run() {
        SessionState.SetBool(Key,true);
        var s=SdkProjectSettings.instance;
        s.gameId="test-game";s.platform=DistributionPlatform.Mock;s.production=false;
        s.login=true;s.payments=false;s.rewardedAds=false;s.share=false;s.storage=false;s.network=false;s.analytics=false;
        s.SaveSettings();
        EditorApplication.isPlaying=true;
    }
    static void Changed(PlayModeStateChange state) {
        if(!SessionState.GetBool(Key,false))return;
        if(state==PlayModeStateChange.EnteredPlayMode) {
            deadline=EditorApplication.timeSinceStartup+20;
            EditorApplication.update+=Step;
            try { EditorSdkBootstrap.Initialize(MockLoginOutcome.Success); SDK.Login(r=>{if(!r.Success)Fail("success failed");else phase=1;}); }
            catch(Exception e){Fail(e.Message);}
        }
    }
    static void Step() {
        if(finished)return;
        if(EditorApplication.timeSinceStartup>deadline){Fail("test deadline");return;}
        try {
            if(phase==1){phase=2;SdkProjectSettings.instance.login=false;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);SDK.Login(r=>{if(r.Error==null||r.Error.Code!="MODULE.DISABLED")Fail("disabled failed");else phase=3;});}
            if(phase==3){phase=4;SdkProjectSettings.instance.login=true;EditorSdkBootstrap.Initialize(MockLoginOutcome.Timeout);SDK.Login(r=>{if(r.Error==null||r.Error.Code!="AUTH.TIMEOUT")Fail("timeout failed");else phase=5;},50);}
            if(phase==5){phase=6;EditorSdkBootstrap.Initialize(MockLoginOutcome.Denied);SDK.Login(r=>{if(r.Error==null||r.Error.Code!="AUTH.PERMISSION_DENIED")Fail("denied failed");else phase=7;});}
            if(phase==7){
                var settings=SdkProjectSettings.instance;
                settings.storage=true;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                if(!SDK.SetStorageSync("play-key","play-value").Success || SDK.GetStorageSync("play-key").Value!="play-value")throw new Exception("storage roundtrip");
                if(!SDK.RemoveStorageSync("play-key").Success || SDK.GetStorageSync("play-key").Error.Code!="STORAGE.NOT_FOUND")throw new Exception("storage removal");
                settings.storage=false;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                if(SDK.GetStorageSync("play-key").Error.Code!="MODULE.DISABLED")throw new Exception("storage disabled");
                settings.share=true;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);phase=8;
                SDK.Share(new ShareRequest("play share"),r=>{if(!r.Success)Fail("share dispatch failed");else phase=9;});
            }
            if(phase==9){
                SdkProjectSettings.instance.share=false;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);phase=10;
                SDK.Share(new ShareRequest("disabled share"),r=>{if(r.Error==null||r.Error.Code!="MODULE.DISABLED")Fail("share disabled failed");else phase=11;});
            }
            if(phase==11){
                phase=12;SdkProjectSettings.instance.payments=true;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                SDK.Pay(new PaymentRequest("mock-session","coins","play-request"),r=>{if(!r.Success)Fail("payment fulfilled failed");else phase=13;});
            }
            if(phase==13){
                phase=14;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success,payment:MockPaymentOutcome.Pending);
                SDK.Pay(new PaymentRequest("mock-session","coins","play-request"),r=>{if(r.Success||r.Error.Code!="PAY.FULFILLMENT_PENDING")Fail("payment pending failed");else phase=15;});
            }
            if(phase==15){
                phase=16;SdkProjectSettings.instance.payments=false;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                SDK.Pay(new PaymentRequest("mock-session","coins","play-request"),r=>{if(r.Error==null||r.Error.Code!="MODULE.DISABLED")Fail("payment disabled failed");else phase=17;});
            }
            if(phase==17){
                phase=18;SdkProjectSettings.instance.rewardedAds=true;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                SDK.ShowRewardedAd(new RewardedAdRequest("mock-unit"),r=>{if(!r.Success||!r.Completed)Fail("ad completed failed");else phase=19;});
            }
            if(phase==19){
                phase=20;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success,ads:MockRewardedAdOutcome.ClosedEarly);
                SDK.ShowRewardedAd(new RewardedAdRequest("mock-unit"),r=>{if(!r.Success||r.Completed)Fail("ad early close failed");else phase=21;});
            }
            if(phase==21){
                phase=22;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success,ads:MockRewardedAdOutcome.LoadFailed);
                SDK.ShowRewardedAd(new RewardedAdRequest("mock-unit"),r=>{if(r.Error==null||r.Error.Code!="ADS.LOAD_FAILED")Fail("ad failure failed");else phase=23;});
            }
            if(phase==23){
                phase=24;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success,ads:MockRewardedAdOutcome.Timeout);
                SDK.ShowRewardedAd(new RewardedAdRequest("mock-unit"),r=>{if(r.Error==null||r.Error.Code!="ADS.TIMEOUT")Fail("ad timeout failed");else phase=25;},50);
            }
            if(phase==25){
                phase=26;SDK.ShowRewardedAd(new RewardedAdRequest("mock-unit"),r=>{if(r.Error==null||r.Error.Code!="ADS.CANCELLED")Fail("ad cancel failed");else phase=27;});SDK.CancelRewardedAd();
            }
            if(phase==27){
                phase=28;SdkProjectSettings.instance.rewardedAds=false;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                SDK.ShowRewardedAd(new RewardedAdRequest("mock-unit"),r=>{if(r.Error==null||r.Error.Code!="MODULE.DISABLED")Fail("ad disabled failed");else phase=29;});
            }
            if(phase==29){
                phase=30;SdkProjectSettings.instance.network=true;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                SDK.Request(new NetworkRequest("https://mock.invalid"),r=>{if(!r.Success||!r.IsHttpSuccess||r.StatusCode!=200)Fail("network 200");else phase=31;});
            }
            if(phase==31){
                phase=32;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success,network:MockNetworkOutcome.HttpUnauthorized);
                SDK.Request(new NetworkRequest("https://mock.invalid"),r=>{if(!r.Success||r.IsHttpSuccess||r.StatusCode!=401)Fail("network 401");else phase=33;});
            }
            if(phase==33){
                phase=34;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success,network:MockNetworkOutcome.ConnectionFailed);
                SDK.Request(new NetworkRequest("https://mock.invalid"),r=>{if(r.Error==null||r.Error.Code!="NETWORK.UNAVAILABLE")Fail("network failure");else phase=35;});
            }
            if(phase==35){
                phase=36;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success,network:MockNetworkOutcome.Timeout);
                SDK.Request(new NetworkRequest("https://mock.invalid",timeoutSeconds:1),r=>{if(r.Error==null||r.Error.Code!="NETWORK.TIMEOUT")Fail("network timeout");else phase=37;});
            }
            if(phase==37){
                phase=38;var operation=SDK.Request(new NetworkRequest("https://mock.invalid"),r=>{if(r.Error==null||r.Error.Code!="NETWORK.CANCELLED")Fail("network cancel");else phase=39;});operation.Cancel();
            }
            if(phase==39){
                phase=40;SdkProjectSettings.instance.network=false;EditorSdkBootstrap.Initialize(MockLoginOutcome.Success);
                SDK.Request(new NetworkRequest("https://mock.invalid"),r=>{if(r.Error==null||r.Error.Code!="MODULE.DISABLED")Fail("network disabled");else phase=41;});
            }
            if(phase==41){
                finished=true;SDK.Shutdown();SessionState.SetBool(Key,false);Debug.Log("SDK_PLAY_CHECKS_PASS");EditorApplication.Exit(0);
            }
        }catch(Exception e){Fail(e.Message);}
    }
    static void Fail(string message){finished=true;SessionState.SetBool(Key,false);Debug.LogError("SDK_PLAY_CHECKS_FAIL: "+message);EditorApplication.Exit(1);}
}
